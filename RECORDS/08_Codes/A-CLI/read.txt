# rename files inside directory say PDF with pdf-o.txt containing file names inside PDF and pdf-n.txt containing new file names

paste pdf-o.txt pdf-n.txt | while IFS=$'\t' read -r o n; do mv "PDF/$o" "PDF/$n"; done
